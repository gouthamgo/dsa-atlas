# DSA Atlas Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A static site that turns a 455-problem DSA curriculum into a dated 90/120/180-day plan with notes, spaced revision and pattern explainers.

**Architecture:** Pure data → pure scheduler → derived schedule joined with a persisted progress store at render time. No server, no database. The schedule is always derived, never stored, so changing plan length cannot corrupt progress.

**Tech Stack:** Next.js 15 (App Router, static export), TypeScript strict, Tailwind CSS v4, Zustand + persist, MDX, Vitest, Playwright.

**Spec:** `docs/superpowers/specs/2026-09-22-dsa-atlas-design.md`

## Global Constraints

- Node 20+. Package manager: npm.
- TypeScript `strict: true`. No `any` in committed code.
- `next.config.ts` sets `output: "export"` — no server-only features (no route handlers, no `next/headers`, no ISR).
- Every localStorage read and write is wrapped in try/catch; the app renders correctly when storage throws.
- Progress is keyed by `Problem["id"]`, never by day index or array position.
- The scheduler is pure: no `Date.now()`, no randomness, no I/O. Dates arrive as ISO strings.
- Dark mode is the default theme; light mode is supported via `data-theme="light"`.
- Commit after each task with a conventional-commit message.

`src/lib/scheduler.ts`:

```ts
import type { Problem, PatternId } from "@/data/types";
import { PATTERN_PREREQS, TOPIC_ORDER } from "@/data/types";

export type Day = {
  index: number;
  date: string;
  problems: Problem[];
  totalMinutes: number;
  isRest: boolean;
};

const DIFFICULTY_RANK = { easy: 0, medium: 1, hard: 2 } as const;

function addDays(iso: string, n: number): string {
  const [y, m, d] = iso.split("-").map(Number);
  const date = new Date(Date.UTC(y, m - 1, d + n));
  return date.toISOString().slice(0, 10);
}

/** Patterns in dependency order, ties broken by the declaration order. */
function orderPatterns(): PatternId[] {
  const out: PatternId[] = [];
  const seen = new Set<PatternId>();
  const visit = (p: PatternId) => {
    if (seen.has(p)) return;
    seen.add(p);
    for (const req of PATTERN_PREREQS[p]) visit(req as PatternId);
    out.push(p);
  };
  for (const p of Object.keys(PATTERN_PREREQS) as PatternId[]) visit(p);
  return out;
}

function sortProblems(problems: Problem[]): Problem[] {
  const patternRank = new Map(orderPatterns().map((p, i) => [p, i]));
  const topicRank = new Map(TOPIC_ORDER.map((t, i) => [t, i]));
  return [...problems].sort((a, b) => {
    const byPattern = patternRank.get(a.pattern)! - patternRank.get(b.pattern)!;
    if (byPattern !== 0) return byPattern;
    const byTopic = topicRank.get(a.topic)! - topicRank.get(b.topic)!;
    if (byTopic !== 0) return byTopic;
    const byDifficulty = DIFFICULTY_RANK[a.difficulty] - DIFFICULTY_RANK[b.difficulty];
    if (byDifficulty !== 0) return byDifficulty;
    return a.id.localeCompare(b.id);
  });
}

export function buildSchedule(input: {
  problems: Problem[];
  days: 90 | 120 | 180;
  startDate: string;
  restEvery?: number;
}): Day[] {
  const restEvery = input.restEvery ?? 7;
  const ordered = sortProblems(input.problems);

  const isRestIndex = (i: number) => restEvery > 0 && (i + 1) % restEvery === 0;
  const workingCount = Array.from({ length: input.days }, (_, i) => i).filter(
    (i) => !isRestIndex(i),
  ).length;

  const totalMinutes = ordered.reduce((s, p) => s + p.minutes, 0);
  const targetPerDay = totalMinutes / Math.max(workingCount, 1);

  const days: Day[] = [];
  let cursor = 0;

  for (let i = 0; i < input.days; i++) {
    const date = addDays(input.startDate, i);
    if (isRestIndex(i)) {
      days.push({ index: i, date, problems: [], totalMinutes: 0, isRest: true });
      continue;
    }

    const remainingWorkingDays = Array.from({ length: input.days - i }, (_, k) => i + k).filter(
      (k) => !isRestIndex(k),
    ).length;
    const remainingProblems = ordered.length - cursor;

    const picked: Problem[] = [];
    let minutes = 0;
    while (cursor < ordered.length) {
      const next = ordered[cursor];
      const wouldExceed = minutes + next.minutes > targetPerDay * 1.2;
      const mustTake = remainingProblems - picked.length >= remainingWorkingDays;
      if (picked.length > 0 && wouldExceed && !mustTake) break;
      picked.push(next);
      minutes += next.minutes;
      cursor++;
      if (minutes >= targetPerDay) break;
    }

    days.push({ index: i, date, problems: picked, totalMinutes: minutes, isRest: false });
  }

  return days;
}
```

import { create } from "zustand";
import { persist, createJSONStorage } from "zustand/middleware";

export type ProblemStatus = "todo" | "solved" | "skipped";
export type Rating = "again" | "hard" | "good";
export type PlanLength = 90 | 120 | 180;

export type Settings = { days: PlanLength; startDate: string; restEvery: number };

export type ProblemProgress = {
  status: ProblemStatus;
  solvedAt?: string;
  rating?: Rating;
  dueAt?: string;
  notes?: string;
  minutesTaken?: number;
};

export type ProgressV1 = {
  version: 1;
  settings: Settings;
  problems: Record<string, ProblemProgress>;
};

const DEFAULTS: Omit<ProgressV1, "version"> = {
  settings: { days: 90, startDate: new Date().toISOString().slice(0, 10), restEvery: 7 },
  problems: {},
};

type Actions = {
  solve: (id: string, minutesTaken?: number) => void;
  skip: (id: string) => void;
  rate: (id: string, rating: Rating, dueAt: string) => void;
  setNote: (id: string, notes: string) => void;
  setDays: (days: PlanLength) => void;
  setStartDate: (startDate: string) => void;
  reset: () => void;
  replaceAll: (next: Omit<ProgressV1, "version">) => void;
};

/** localStorage that never throws — private mode and blocked storage must not break the app. */
const safeStorage = createJSONStorage(() => ({
  getItem: (name: string) => {
    try {
      return localStorage.getItem(name);
    } catch {
      return null;
    }
  },
  setItem: (name: string, value: string) => {
    try {
      localStorage.setItem(name, value);
    } catch {
      /* progress is in memory only this session */
    }
  },
  removeItem: (name: string) => {
    try {
      localStorage.removeItem(name);
    } catch {
      /* nothing to do */
    }
  },
}));

export const useProgress = create<Omit<ProgressV1, "version"> & Actions>()(
  persist(
    (set, get) => ({
      ...DEFAULTS,
      solve: (id, minutesTaken) =>
        set((s) => ({
          problems: {
            ...s.problems,
            [id]: {
              ...s.problems[id],
              status: "solved",
              solvedAt: new Date().toISOString().slice(0, 10),
              minutesTaken,
            },
          },
        })),
      skip: (id) =>
        set((s) => ({
          problems: { ...s.problems, [id]: { ...s.problems[id], status: "skipped" } },
        })),
      rate: (id, rating, dueAt) =>
        set((s) => ({
          problems: { ...s.problems, [id]: { ...s.problems[id], status: s.problems[id]?.status ?? "solved", rating, dueAt } },
        })),
      setNote: (id, notes) =>
        set((s) => ({
          problems: { ...s.problems, [id]: { ...s.problems[id], status: s.problems[id]?.status ?? "todo", notes } },
        })),
      setDays: (days) => set((s) => ({ settings: { ...s.settings, days } })),
      setStartDate: (startDate) => set((s) => ({ settings: { ...s.settings, startDate } })),
      reset: () => set({ ...DEFAULTS, problems: {} }),
      replaceAll: (next) => set({ settings: next.settings, problems: next.problems }),
    }),
    {
      name: "dsa-atlas",
      version: 1,
      storage: safeStorage,
      partialize: (s) => ({ settings: s.settings, problems: s.problems }),
    },
  ),
);

export function exportProgress(): string {
  const { settings, problems } = useProgress.getState();
  return JSON.stringify({ version: 1, settings, problems }, null, 2);
}

export function importProgress(json: string): { ok: true } | { ok: false; error: string } {
  let parsed: unknown;
  try {
    parsed = JSON.parse(json);
  } catch {
    return { ok: false, error: "That file is not valid JSON." };
  }
  const data = parsed as Partial<ProgressV1>;
  if (data?.version !== 1 || typeof data.settings !== "object" || typeof data.problems !== "object") {
    return { ok: false, error: "That file is not a DSA Atlas export." };
  }
  useProgress.getState().replaceAll({ settings: data.settings as Settings, problems: data.problems as Record<string, ProblemProgress> });
  return { ok: true };
}
```

import type { Rating, ProblemProgress } from "@/lib/store";

export const REVIEW_INTERVALS: Record<Rating, number> = { again: 3, hard: 7, good: 21 };

function addDays(iso: string, n: number): string {
  const [y, m, d] = iso.split("-").map(Number);
  return new Date(Date.UTC(y, m - 1, d + n)).toISOString().slice(0, 10);
}

export function nextDueDate(rating: Rating, fromISO: string): string {
  return addDays(fromISO, REVIEW_INTERVALS[rating]);
}

/** Ids due on or before today, oldest first, capped so revision never buries new work. */
export function dueProblems(
  problems: Record<string, ProblemProgress>,
  todayISO: string,
  cap = 5,
): string[] {
  return Object.entries(problems)
    .filter(([, p]) => p.dueAt !== undefined && p.dueAt <= todayISO)
    .sort((a, b) => a[1].dueAt!.localeCompare(b[1].dueAt!) || a[0].localeCompare(b[0]))
    .slice(0, cap)
    .map(([id]) => id);
}
```

"use client";

import { useMemo } from "react";
import { buildSchedule, type Day } from "@/lib/scheduler";
import { ALL_PROBLEMS } from "@/data/problems";
import { useProgress, type Settings, type ProblemProgress } from "@/lib/store";
import type { Problem } from "@/data/types";

export type ScheduleView = { schedule: Day[]; today: Day | null; behindBy: number };

export function computeView(
  problems: Problem[],
  settings: Settings,
  progress: Record<string, ProblemProgress>,
  todayISO: string,
): ScheduleView {
  const schedule = buildSchedule({
    problems,
    days: settings.days,
    startDate: settings.startDate,
    restEvery: settings.restEvery,
  });
  const today = schedule.find((d) => d.date === todayISO) ?? null;
  const behindBy = schedule
    .filter((d) => d.date < todayISO)
    .flatMap((d) => d.problems)
    .filter((p) => progress[p.id]?.status !== "solved" && progress[p.id]?.status !== "skipped").length;
  return { schedule, today, behindBy };
}

export function useSchedule(): ScheduleView {
  const settings = useProgress((s) => s.settings);
  const problems = useProgress((s) => s.problems);
  const todayISO = new Date().toISOString().slice(0, 10);
  return useMemo(
    () => computeView(ALL_PROBLEMS, settings, problems, todayISO),
    [settings, problems, todayISO],
  );
}
